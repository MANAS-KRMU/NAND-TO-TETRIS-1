Memory.hdl
CHIP Memory {
    IN in[16], load, address[15];
    OUT out[16];
    PARTS:
    // 00, 01 -> RAM16K (0..16383)
    // 10     -> Screen (16384..24575)
    // 11     -> Keyboard (24576)
    DMux4Way(in=load, sel=address[13..14],
             a=loadRamA, b=loadRamB,
             c=loadScreen, d=loadKbd);
    Or(a=loadRamA, b=loadRamB, out=loadRam);
    RAM16K(in=in, load=loadRam,
           address=address[0..13], out=ramOut);
    Screen(in=in, load=loadScreen,
           address=address[0..12], out=scrOut);
    Keyboard(out=kbdOut);
    Mux4Way16(a=ramOut, b=ramOut,
              c=scrOut, d=kbdOut,
              sel=address[13..14], out=out);
}
CPU.hdl (Control Logic)
CHIP CPU {
    IN inM[16], instruction[16], reset;
    OUT outM[16], writeM, addressM[15], pc[15];
    PARTS:
    // Decode A vs C instruction (bit 15)
    Not(in=instruction[15], out=isAInst);
    Mux16(a=aluOut, b=instruction,
          sel=isAInst, out=aRegIn);
    Or(a=isAInst, b=instruction[5], out=loadA);
    ARegister(in=aRegIn, load=loadA,
              out=aRegOut, out[0..14]=addressM);
    // D Register & ALU
    And(a=instruction[15], b=instruction[4], out=loadD);
    DRegister(in=aluOut, load=loadD, out=dRegOut);
    Mux16(a=aRegOut, b=inM,
          sel=instruction[12], out=aluY);
    ALU(x=dRegOut, y=aluY,
        zx=instruction[11], nx=instruction[10],
        zy=instruction[9],  ny=instruction[8],
        f=instruction[7],   no=instruction[6],
        out=aluOut, out=outM, zr=zr, ng=ng);
    And(a=instruction[15], b=instruction[3], out=writeM);
}

Quick Viva Demonstration Cheatsheet

If your examiner asks you to modify the box position or size live during the 5-minute viva:

  1. Move the box down by 10 rows: Change @18784 in both (DRAW_BOX) and (ERASE_BOX) to @19104 (18784 + 32 × 10), then click Assemble & Load ROM.
  2. Change box height: Change @20 to the requested row count (e.g., @30).
  3. Change box width: Change @4 (words per row) to @2 for 32 pixels or @6 for 96 pixels.
  4. Or simply enter a new K in the top bar and click Recalculate & Load Box.asm to see the updated assembly code immediately!
